738 template<typename T>
739 std::shared_ptr<T> ReadExtendedField(const std::string& label,
740 const Extended::Item* item) const
741 {
742 std::shared_ptr<T> field;
743
744 std::string extendedLabel = label + "*";
745
746 if (item != nullptr)
747 {
748 if (item->type->ToInt32() == Extended::lengthType)
749 {
750 // Copy the raw data to a new field to ensure it is
751 // interpreted according to the correct type T as item->data
752 // is created as a NumericField, but may represent a
753 // BinaryField or other type and needs to be recreated as
754 // such.
755 field = std::make_shared<T>(extendedLabel,
756 Extended::dataInfo);
757 item->data->CopyRawDataTo(*field);
758 }
759 else if (item->extendedData != nullptr)
760 {
761 field = std::static_pointer_cast<T>(item->extendedData);
762 field->SetLabel(extendedLabel);
763 }
764 else
765 {
766 field = std::make_shared<T>(extendedLabel,
767 Extended::dataInfo);
768 }
769 }
770 else
771 {
772 field = std::make_shared<T>(extendedLabel,
773 Extended::dataInfo,
774 false);
775 }
776
777 return field;
778 }

902 template<typename T>
903 void WriteExtendedTextField(
904 const TagFieldInfo& info,
905 Extended::ItemInfo extendedInfo,
906 std::shared_ptr<Extended::Item>* itemPtrPtr,
907 const std::string& value)
908 {
909 if (value.empty())
910 {
911 WriteTextField<T>(info, value);
912 *(itemPtrPtr) = nullptr;
913 return;
914 }
915
916 size_t fieldSize;
917
918 if (DetermineType() == TagType::Binary)
919 {
920 fieldSize = info.binary.size;
921 }
922 else
923 {
924 fieldSize = info.text.size;
925 }
926
927 if (value.size() > fieldSize)
928 {
929 WriteExtendedStringField<T>(extendedInfo,
930 itemPtrPtr,
931 value);
932 WriteTextField<T>(info, value);
933 }
934 else
935 {
936 *(itemPtrPtr) = nullptr;
937 WriteTextField<T>(info, value);
938 }

1076 template<typename T>
1077 void WriteExtendedStringField(
1078 Extended::ItemInfo extendedInfo,
1079 std::shared_ptr<Extended::Item>* itemPtrPtr,
1080 const std::string& value)
1081 {
1082 if (value.empty())
1083 {
1084 *(itemPtrPtr) = nullptr;
1085 return;
1086 }
1087
1088 if (value.size() > maxStringSize)
1089 {
1090 throw std::out_of_range(
1091 "Value exceeds maximum string size for extended data.");
1092 }
1093
1094 if (*(itemPtrPtr) == nullptr)
1095 {
1096 auto item = std::make_shared<Extended::Item>();
1097 item->id->SetInt32(extendedInfo.id);
1098 item->type->SetInt32(extendedInfo.type);
1099
1100 auto data = std::static_pointer_cast<NumericField>(item->data);
1101
1102 data->SetUInt32(static_cast<uint32_t>(value.size()));
1103 Spc::FieldInfo info{ Extended::dataOffset, value.size() };
1104 auto field = std::make_shared<T>("Temp", info);
1105 field->SetValue(value);
1106 item->extendedData = field;
1107
1108 if (value.size() % 4 != 0)
1109 {
1110 size_t paddingSize{ 0 };
1111
1112 while ((value.size() + paddingSize) % 4 != 0)
1113 {
1114 paddingSize++;
1115 }
1116
1117 FieldInfo paddingInfo{ Extended::dataOffset, paddingSize };
1118 item->padding = std::make_shared<TextField>("Padding",
1119 paddingInfo);
1120 }
1121 else
1122 {
1123 item->padding = nullptr;
1124 }
1125
1126 *(itemPtrPtr) = item;
1127 }
1128 else
1129 {
1130 auto item = *(itemPtrPtr);
1131
1132 auto data = std::static_pointer_cast<NumericField>(item->data);
1133
1134 data->SetUInt32(static_cast<uint32_t>(value.size()));
1135 Spc::FieldInfo info{ Extended::dataOffset, value.size() };
1136 item->extendedData = std::make_shared<T>("Temp", info);
1137 item->extendedData->SetValue(value);
1138
1139 if (value.size() % 4 != 0)
1140 {
1141 size_t paddingSize{ 0 };
1142
1143 while ((value.size() + paddingSize) % 4 != 0)
1144 {
1145 paddingSize++;
1146 }
1147
1148 FieldInfo paddingInfo{ Extended::dataOffset, paddingSize };
1149 item->padding = std::make_shared<TextField>("Padding",
1150 paddingInfo);
1151 }
1152 else
1153 {
1154 item->padding = nullptr;
1155 }
1156 }
1157 }
