701 template<typename T>
702 std::shared_ptr<T> ReadExtendedField(std::string label,
703 Extended::Item* item) const
704 {
705 std::shared_ptr<T> field;
706
707 if (item != nullptr)
708 {
709 if (item->type->ToInt32() == Extended::lengthType)
710 {
711 // Copy the raw data to a new field to ensure it is
712 // interpreted according to the correct type T as item->data
713 // is created as a NumericField, but may represent a
714 // BinaryField or other type and needs to be recreated as
715 // such.
716 field = std::make_shared<T>(label, Extended::dataInfo);
717 item->data->CopyRawDataTo(field.get());
718 }
719 else if (item->extendedData != nullptr)
720 {
721 field = std::static_pointer_cast<T>(item->extendedData);
722 }
723 else
724 {
725 field = std::make_shared<T>(label, Extended::dataInfo);
726 }
727 }
728 else
729 {
730 field = std::make_shared<T>(label, Extended::dataInfo);
731 }
732
733 return field;
734 }

922 {

924 {

926 item->id->SetInt32(extendedInfo.id);
927 item->type->SetInt32(extendedInfo.type);
928

930 data->SetInt32(Extended::integerSize);
931
932 Spc::FieldInfo info;
933 info.offset = Extended::dataOffset;
934 info.size = Extended::integerSize;
935 auto field = std::make_shared<T>("Temp", info);
936 field->SetType(Spc::NumericType::Binary);
937 field->SetValue(value);
938 item->extendedData = field;
939
940 *(itemPtrPtr) = item;
941 }
942 else

944 auto item = *(itemPtrPtr);
945
946 std::shared_ptr<NumericField> data = item->data;
947
948 data->SetInt32(Extended::integerSize);
949 Spc::FieldInfo info;

953

955 extData->SetType(Spc::NumericType::Binary);

977 template<typename T>
978 void WriteExtendedStringField(
979 Extended::ItemInfo extendedInfo,
980 std::shared_ptr<Extended::Item>* itemPtrPtr,
981 std::string value)
982 {
983 if (value.size() > maxStringSize)
984 {
985 throw std::out_of_range(
986 "Value exceeds maximum string size for extended data.");
987 }
988
989 if (*(itemPtrPtr) == nullptr)
990 {
991 auto item = std::make_shared<Extended::Item>();
992 item->id->SetInt32(extendedInfo.id);
993 item->type->SetInt32(extendedInfo.type);
994
995 auto data = std::static_pointer_cast<NumericField>(item->data);
996
997 data->SetUInt32(static_cast<uint32_t>(value.size()));
998 Spc::FieldInfo info{ Extended::dataOffset, value.size() };
999 auto field = std::make_shared<T>("Temp", info);
1000 field->SetValue(value);
1001 item->extendedData = field;
1002
1003 *(itemPtrPtr) = item;
1004 }
1005 else
1006 {
1007 auto item = *(itemPtrPtr);
1008
1009 auto data = std::static_pointer_cast<NumericField>(item->data);
1010
1011 data->SetUInt32(static_cast<uint32_t>(value.size()));
1012 Spc::FieldInfo info{ Extended::dataOffset, value.size() };
1013 item->extendedData = std::make_shared<T>("Temp", info);
1014 item->extendedData->SetValue(value);
1015 }
1016 }
