764 template<typename T>
765 std::shared_ptr<T> ReadExtendedField(const std::string& label,
766 const Extended::Item* item) const
767 {
768 std::shared_ptr<T> field;
769
770 std::string extendedLabel = label + "*";
771
772 if (item != nullptr)
773 {
774 if (item->type->ToInt32() == Extended::lengthType)
775 {
776 // Copy the raw data to a new field to ensure it is
777 // interpreted according to the correct type T as item->data
778 // is created as a NumericField, but may represent a
779 // BinaryField or other type and needs to be recreated as
780 // such.
781 field = std::make_shared<T>(extendedLabel,
782 Extended::dataInfo);
783 item->data->CopyRawDataTo(*field);
784 }
785 else if (item->extendedData != nullptr)
786 {
787 field = std::static_pointer_cast<T>(item->extendedData);
788 field->SetLabel(extendedLabel);
789 }
790 else
791 {
792 field = std::make_shared<T>(extendedLabel,
793 Extended::dataInfo);
794 }
795 }
796 else
797 {
798 field = std::make_shared<T>(extendedLabel,
799 Extended::dataInfo,
800 false);
801 }
802
803 return field;
804 }

929 void WriteExtendedTextField(
930 const TagFieldInfo& info,
931 Extended::ItemInfo extendedInfo,
932 std::shared_ptr<Extended::Item>* itemPtrPtr,
933 const std::string& value)
934 {
935 if (value.empty())
936 {
937 WriteTextField<T>(info, value);
938 *(itemPtrPtr) = nullptr;
939 return;
940 }
941
942 size_t fieldSize;
943
944 if (DetermineType() == TagType::Binary)
945 {
946 fieldSize = info.binary.size;
947 }
948 else
949 {
950 fieldSize = info.text.size;
951 }
952
953 if (value.size() > fieldSize)
954 {
955 WriteExtendedStringField<T>(extendedInfo,
956 itemPtrPtr,
957 value);
958 WriteTextField<T>(info, value);
959 }
960 else
961 {
962 *(itemPtrPtr) = nullptr;
963 WriteTextField<T>(info, value);

1102 template<typename T>
1103 void WriteExtendedStringField(
1104 Extended::ItemInfo extendedInfo,
1105 std::shared_ptr<Extended::Item>* itemPtrPtr,
1106 const std::string& value)
1107 {
1108 if (value.empty())
1109 {
1110 *(itemPtrPtr) = nullptr;
1111 return;
1112 }
1113
1114 if (value.size() > maxStringSize)
1115 {
1116 throw std::out_of_range(
1117 "Value exceeds maximum string size for extended data.");
1118 }
1119
1120 if (*(itemPtrPtr) == nullptr)
1121 {
1122 auto item = std::make_shared<Extended::Item>();
1123 item->id->SetInt32(extendedInfo.id);
1124 item->type->SetInt32(extendedInfo.type);
1125
1126 auto data = std::static_pointer_cast<NumericField>(item->data);
1127
1128 data->SetUInt32(static_cast<uint32_t>(value.size()));
1129 Spc::FieldInfo info{ Extended::dataOffset, value.size() };
1130 auto field = std::make_shared<T>("Temp", info);
1131 field->SetValue(value);
1132 item->extendedData = field;
1133
1134 if (value.size() % 4 != 0)
1135 {
1136 size_t paddingSize{ 0 };
1137
1138 while ((value.size() + paddingSize) % 4 != 0)
1139 {
1140 paddingSize++;
1141 }
1142
1143 FieldInfo paddingInfo{ Extended::dataOffset, paddingSize };
1144 item->padding = std::make_shared<TextField>("Padding",
1145 paddingInfo);
1146 }
1147 else
1148 {
1149 item->padding = nullptr;
1150 }
1151
1152 *(itemPtrPtr) = item;
1153 }
1154 else
1155 {
1156 auto item = *(itemPtrPtr);
1157
1158 auto data = std::static_pointer_cast<NumericField>(item->data);
1159
1160 data->SetUInt32(static_cast<uint32_t>(value.size()));
1161 Spc::FieldInfo info{ Extended::dataOffset, value.size() };
1162 item->extendedData = std::make_shared<T>("Temp", info);
1163 item->extendedData->SetValue(value);
1164
1165 if (value.size() % 4 != 0)
1166 {
1167 size_t paddingSize{ 0 };
1168
1169 while ((value.size() + paddingSize) % 4 != 0)
1170 {
1171 paddingSize++;
1172 }
1173
1174 FieldInfo paddingInfo{ Extended::dataOffset, paddingSize };
1175 item->padding = std::make_shared<TextField>("Padding",
1176 paddingInfo);
1177 }
1178 else
1179 {
1180 item->padding = nullptr;
1181 }
1182 }
1183 }
